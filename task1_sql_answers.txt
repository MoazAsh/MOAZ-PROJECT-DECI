# Task 1 - Step 4: SQL JOIN
# ==========================================

sql_query = """
SELECT
    c.checkout_id,
    c.member_id,
    m.first_name,
    m.last_name,
    m.grade,
    m.neighborhood,
    m.membership_status,
    m.join_date,
    c.book_id,
    c.checkout_date,
    c.return_date
FROM checkouts AS c
INNER JOIN members AS m
    ON c.member_id = m.member_id
WHERE c.member_id IS NOT NULL
"""

checkout_member_data = pd.read_sql_query(sql_query, conn)

print("Records after SQL JOIN:", len(checkout_member_data))
display(checkout_member_data.head())
